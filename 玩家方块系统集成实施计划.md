#### 5.2 创建 BlockDropListener.cpp

**文件**: [Classes/systems/block/BlockDropListener.cpp](Classes/systems/block/BlockDropListener.cpp)

```cpp
#include "BlockDropListener.h"
#include "core/GameManager.h"
#include "systems/items/Inventory.h"
#include "systems/items/ItemManager.h"
#include <unordered_map>

void BlockDropListener::initialize(entt::dispatcher& dispatcher) {
    dispatcher.sink<BlockDestroyEvent>().connect<&BlockDropListener::onBlockDestroyed>();
    CCLOG("[BlockDropListener] Initialized");
}

void BlockDropListener::cleanup(entt::dispatcher& dispatcher) {
    dispatcher.sink<BlockDestroyEvent>().disconnect<&BlockDropListener::onBlockDestroyed>();
    CCLOG("[BlockDropListener] Cleaned up");
}

void BlockDropListener::onBlockDestroyed(const BlockDestroyEvent& event) {
    // 获取掉落物品ID
    int droppedItemId = getDroppedItemFromBlock(event.id);
    if (droppedItemId == 0) {
        CCLOG("[BlockDropListener] Block %d has no drop", event.id);
        return;
    }

    int droppedCount = 1; // 默认掉落1个

    // 检查破坏者是否是玩家
    auto playerEntity = GameManager::getInstance()->getPlayerEntity();
    if (event.breaker != playerEntity) {
        CCLOG("[BlockDropListener] Breaker is not player, creating drop entity instead");
        // TODO: 在世界中生成掉落物实体
        return;
    }

    // 添加到玩家背包
    auto overflow = Inventory::getInstance()->addItemWithOverflow(droppedItemId, droppedCount);

    if (overflow.empty()) {
        CCLOG("[BlockDropListener] Added %d x %d to player inventory",
              droppedItemId, droppedCount);
    } else {
        CCLOG("[BlockDropListener] Inventory full! %d items overflow",
              static_cast<int>(overflow.size()));
        // TODO: 在世界中生成掉落物实体
    }
}

int BlockDropListener::getDroppedItemFromBlock(entt::id_type blockId) {
    // 方块ID → 掉落物品ID 映射表
    static std::unordered_map<entt::id_type, int> blockToItemMap = {
        {1, 2001},   // 泥土方块 → 泥土物品
        {2, 2002},   // 石头方块 → 石头物品
        {3, 2003},   // 木头方块 → 木头物品
        // 继续添加其他方块...
    };

    auto it = blockToItemMap.find(blockId);
    return (it != blockToItemMap.end()) ? it->second : 0;
}
```

**文件**: [Classes/systems/player/ItemUseSystem.cpp](Classes/systems/player/ItemUseSystem.cpp)

```cpp
#include "ItemUseSystem.h"
#include "core/GameManager.h"
#include "systems/items/ItemManager.h"
#include "systems/items/Inventory.h"
#include "PlayerAnimationSystem.h"
#include "components/player/PlayerComponents.h"

bool ItemUseSystem::placeBlock(entt::registry& registry,
                                entt::entity playerEntity,
                                int itemId,
                                cocos2d::Vec2 worldPos) {
    // 1. 获取物品数据
    auto* itemData = ItemManager::getInstance()->getItemData(itemId);
    if (!itemData) {
        CCLOG("[ItemUseSystem] Invalid item ID: %d", itemId);
        return false;
    }

    // 2. 检查是否可放置
    if (!itemData->placeable) {
        CCLOG("[ItemUseSystem] Item %s is not placeable", itemData->name.c_str());
        return false;
    }

    // 3. 获取 BlockWorld
    auto& blockWorld = GameManager::getInstance()->getBlockWorld();

    // 4. 检查该位置是否已有方块
    BlockHandle currentBlock = blockWorld.getBlockAtWorldPos(LayerType::BLOCK, worldPos);
    if (currentBlock.isIDVailed()) {
        CCLOG("[ItemUseSystem] Position already has a block");
        return false;
    }

    // 5. 尝试放置方块
    bool success = blockWorld.tryPlaceAtWorldPos(
        worldPos,
        itemData->tileId,  // 物品对应的方块ID
        0,                 // 默认状态
        playerEntity
    );

    if (!success) {
        CCLOG("[ItemUseSystem] Failed to place block");
        return false;
    }

    // 6. 消耗背包物品
    if (!Inventory::getInstance()->removeItem(itemId, 1)) {
        CCLOG("[ItemUseSystem] Failed to remove item from inventory");
        // 注意：方块已经放置，但背包移除失败（理论上不应该发生）
        return false;
    }

    // 7. 播放放置动画
    auto* anim = registry.try_get<ecs::PlayerAnimationComponent>(playerEntity);
    if (anim) {
        PlayerAnimationSystem::transitionToState(*anim,
            ecs::PlayerAnimationComponent::AnimState::PLACE);
    }

    CCLOG("[ItemUseSystem] Placed block %d (item: %s) at (%.2f, %.2f)",
          itemData->tileId, itemData->name.c_str(), worldPos.x, worldPos.y);

    return true;
}
```

```cpp
void PlayerInputSystem::handleMining(entt::registry& registry, float dt) {
    auto& input = PlayerInput::getInstance();

    // 检测左键按住
    if (!input.isMousePressed(cocos2d::EventMouse::MouseButton::BUTTON_LEFT)) {
        return;
    }

    // 获取鼠标世界坐标
    cocos2d::Vec2 mouseWorldPos = input.getMouseWorldPosition();

    // 获取玩家实体
    auto playerEntity = GameManager::getInstance()->getPlayerEntity();
    if (playerEntity == entt::null) return;

    // 获取玩家快捷栏选中物品
    auto* hotbar = registry.try_get<ecs::PlayerHotbarComponent>(playerEntity);
    if (!hotbar) return;

    int inventoryIndex = hotbar->getCurrentInventoryIndex();
    auto& inventory = Inventory::getInstance();
    auto slot = inventory->getSlots()[inventoryIndex];

    // 计算挖掘强度
    float mineFactor = 1.0f; // 默认徒手挖掘

    if (slot.itemId > 0) {
        auto* itemData = ItemManager::getInstance()->getItemData(slot.itemId);
        if (itemData && itemData->equipType == EquipType::Pickaxe) {
            // 镐子挖掘强度 = 基础值 + 伤害值 * 系数
            mineFactor = 5.0f + itemData->damage * 0.5f;

            // 播放镐子动画
            auto* anim = registry.try_get<ecs::PlayerAnimationComponent>(playerEntity);
            if (anim && anim->currentState != ecs::PlayerAnimationComponent::AnimState::PICKAXE) {
                PlayerAnimationSystem::transitionToState(*anim,
                    ecs::PlayerAnimationComponent::AnimState::PICKAXE);
            }
        }
    }

    // 调用挖掘接口
    auto& blockWorld = GameManager::getInstance()->getBlockWorld();
    bool success = blockWorld.tryMineAtWorldPos(
        LayerType::BLOCK,
        mouseWorldPos,
        mineFactor * dt,  // 每帧累积进度
        playerEntity
    );

    if (success) {
        // 挖掘成功（进度增加）
        // BlockMiningSystem 会自动显示挖掘进度
    }
}
```

```cpp
void PlayerInputSystem::handlePlacement(entt::registry& registry) {
    auto& input = PlayerInput::getInstance();

    // 检测右键单击
    if (!input.isMouseJustPressed(cocos2d::EventMouse::MouseButton::BUTTON_RIGHT)) {
        return;
    }

    // 获取鼠标世界坐标
    cocos2d::Vec2 mouseWorldPos = input.getMouseWorldPosition();

    // 获取玩家实体
    auto playerEntity = GameManager::getInstance()->getPlayerEntity();
    if (playerEntity == entt::null) return;

    // 首先尝试交互方块（优先级更高）
    auto& blockWorld = GameManager::getInstance()->getBlockWorld();
    bool interacted = blockWorld.tryInteractAtWorldPos(
        LayerType::BLOCK,
        mouseWorldPos,
        playerEntity
    );

    if (interacted) {
        CCLOG("[PlayerInput] Interacted with block");
        return; // 交互成功，不再尝试放置
    }

    // 没有交互到方块，尝试放置
    auto* hotbar = registry.try_get<ecs::PlayerHotbarComponent>(playerEntity);
    if (!hotbar) return;

    int inventoryIndex = hotbar->getCurrentInventoryIndex();
    auto& inventory = Inventory::getInstance();
    auto slot = inventory->getSlots()[inventoryIndex];

    if (slot.itemId <= 0) {
        return; // 没有选中物品
    }

    // 调用放置方块
    bool placed = ItemUseSystem::placeBlock(
        registry,
        playerEntity,
        slot.itemId,
        mouseWorldPos
    );

    if (placed) {
        CCLOG("[PlayerInput] Placed block at (%.2f, %.2f)",
              mouseWorldPos.x, mouseWorldPos.y);
    }
}
```

```cpp
void PlayerInputSystem::detectNearbyWorkbenches(entt::registry& registry) {
    auto playerEntity = GameManager::getInstance()->getPlayerEntity();
    if (playerEntity == entt::null) return;

    auto* transform = registry.try_get<ecs::TransformComponent>(playerEntity);
    if (!transform) return;

    cocos2d::Vec2 playerPos(transform->x, transform->y);
    auto& blockWorld = GameManager::getInstance()->getBlockWorld();

    // 清除之前的工作台检测
    StationDetector::getInstance()->clear();

    // 检测周围5格内的方块
    const int DETECT_RADIUS = 5;
    for (int dx = -DETECT_RADIUS; dx <= DETECT_RADIUS; dx++) {
        for (int dy = -DETECT_RADIUS; dy <= DETECT_RADIUS; dy++) {
            cocos2d::Vec2 checkPos = playerPos +
                cocos2d::Vec2(dx * BLOCK_SIZE, dy * BLOCK_SIZE);

            BlockHandle block = blockWorld.getBlockAtWorldPos(
                LayerType::BLOCK,
                checkPos
            );

            if (block.isIDVailed()) {
                // 根据方块ID判断工作台类型
                switch (block.id.value()) {
                    case 100: // 假设100是工作台方块ID
                        StationDetector::getInstance()->addStation(StationType::Workbench);
                        break;
                    case 101: // 假设101是熔炉方块ID
                        StationDetector::getInstance()->addStation(StationType::Furnace);
                        break;
                    case 102: // 假设102是铁砧方块ID
                        StationDetector::getInstance()->addStation(StationType::Anvil);
                        break;
                }
            }
        }
    }
}
```

在 `update()` 中每隔一定时间调用：

```cpp
void PlayerInputSystem::update(entt::registry& registry, float dt) {
    // ... 原有逻辑 ...

    // 每0.5秒检测一次工作台
    static float detectTimer = 0.0f;
    detectTimer += dt;
    if (detectTimer >= 0.5f) {
        detectNearbyWorkbenches(registry);
        detectTimer = 0.0f;
    }
}
```
